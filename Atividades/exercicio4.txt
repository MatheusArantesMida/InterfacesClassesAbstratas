using System;
using System.Collections.Generic;

namespace InterfacesClassesAbstratas
{
    internal class Program
    {
        static void Main()
        {
            List<IValidavel> campos = new List<IValidavel>
        {
            new CampoTexto
            {
                NomeDoCampo = "Nome",
                Valor = "Matheus"
            },
            new CampoTexto
            {
                NomeDoCampo = "Sobrenome",
                Valor = ""
            },
            new CampoNumerico
            {
                NomeDoCampo = "Idade",
                Valor = "ABC"
            }
        };
            foreach (IValidavel item in campos)
            {
                bool valido = item.Validar();

                Console.WriteLine(
                    $"{campo.NomeDoCampo}: " +
                    $"{(valido ? "Válido" : "Inválido")}");
            }
        }
    }
}

namespace InterfacesClassesAbstratas
{
    internal interface IValidavel
    {
        bool Validar();
    }
}

namespace InterfacesClassesAbstratas
{
    internal abstract class CampoFormulario
    {
        public string NomeDoCampo { get; set; } = "";
        public string Valor { get; set; } = "";
    }
}

namespace InterfacesClassesAbstratas
{
    internal class CampoTexto : CampoFormulario, IValidavel
    {
        public bool Validar()
        {
            return !string.IsNullOrEmpty(Valor);
        }
    }
}

namespace InterfacesClassesAbstratas
{
    internal class CampoNumerico : CampoFormulario, IValidavel
    {
        public bool Validar()
        {
            return int.TryParse(Valor, out _);
        }
    }
}
